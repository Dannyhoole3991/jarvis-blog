Step 4 — Copy this Apps Script into your sheet (one paste, no coding required)

Open Extensions → Apps Script and paste this short script. It grabs a draft template, replaces placeholders like {{Name}} and {{Date}}, and creates drafts or sends messages.

Apps Script (paste into the editor)

function mailMergeFromDraft() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Sheet1'); // change if needed
  const data = sheet.getDataRange().getValues();
  const headers = data.shift();
  const emailIdx = headers.indexOf('Email');
  const statusIdx = headers.indexOf('Status');

  // Find the Gmail draft by subject fragment (change to match your template name)
  const drafts = GmailApp.getDrafts();
  let templateBody = null;
  for (let d of drafts) {
    const msg = d.getMessage();
    if (msg.getSubject().includes('Appt confirm')) { // use part of your template subject
      templateBody = msg.getBody();
      var templateSubject = msg.getSubject();
      break;
    }
  }
  if (!templateBody) {
    SpreadsheetApp.getUi().alert('Template draft not found. Make sure you have a draft with "Appt confirm" in the subject.');
    return;
  }

  // Process each row
  for (let i = 0; i < data.length; i++) {
    const row = data[i];
    if (row[statusIdx] && row[statusIdx].toString().toLowerCase() === 'sent') continue;
    const email = row[emailIdx];
    if (!email) continue;

    // Build replacement map from columns
    let filledBody = templateBody;
    for (let c = 0; c < headers.length; c++) {
      const key = '{{' + headers[c] + '}}';
      const value = row[c] || '';
      const re = new RegExp(key, 'g');
      filledBody = filledBody.replace(re, value);
    }

    // Create draft (or use GmailApp.sendEmail to send immediately)
    GmailApp.createDraft(email, templateSubject.replace(/{{.*?}}/g, ''), '', {htmlBody: filledBody});
    sheet.getRange(i+2, statusIdx+1).setValue('Draft created'); // mark row (i+2 because header row removed)
    Utilities.sleep(500); // small delay for safety
  }
}

Notes: - Change the subject fragment ('Appt confirm') and sheet name as needed. - This script creates drafts (safer). To send immediately, replace GmailApp.createDraft(...) with GmailApp.sendEmail(...) and appropriate parameters. - You’ll be asked to authorize the script the first time. Apps Script uses Gmail and Sheets scopes. (developers.google.com)
